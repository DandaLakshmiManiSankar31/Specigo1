from flask import Flask, request, jsonify
from flask_cors import CORS
from transformers import AutoProcessor, AutoModelForImageTextToText
import torch
from collections import defaultdict
import threading
from pyngrok import ngrok
import time

# -------------------------
# Flask setup
# -------------------------
app = Flask(__name__)
CORS(app)

# -------------------------
# Globals
# -------------------------
model = None
processor = None
conversation_sessions = defaultdict(list)
session_lock = threading.Lock()

MODEL_ID = "google/medgemma-4b-it"

print("=" * 70)
print("🏥 MEDGEMMA MEDICAL ASSISTANT API")
print("=" * 70)

# -------------------------
# Load model
# -------------------------
def setup_model():
    global model, processor

    try:
        if model is None:
            print("📦 Loading MedGemma model...")
            processor = AutoProcessor.from_pretrained(MODEL_ID)
            model = AutoModelForImageTextToText.from_pretrained(
                MODEL_ID,
                device_map="auto",
                # quantization_config={"load_in_4bit": True},
                torch_dtype=torch.bfloat16,
            )
        print("✅ Model loaded")
        print(f"🎯 Device: {model.device}")
        return True
    except Exception as e:
        print(f"❌ Model load failed: {e}")
        return False


# -------------------------
# Health check
# -------------------------
@app.route("/health", methods=["GET"])
def health():
    return jsonify({
        "status": "healthy",
        "model_loaded": model is not None,
        "active_sessions": len(conversation_sessions),
        "device": str(model.device) if model else "N/A"
    })


# -------------------------
# Start session
# -------------------------
@app.route("/start_session", methods=["POST"])
def start_session():
    data = request.json or {}
    session_id = str(data.get("session_id") or data.get("userId") or "default")
    patient_info = data.get("context") or data.get("patient_info", "")

    instruction = (
        "Instruction: You are a medical triage doctor. "
        "Ask ONE follow-up question per turn. "
        "Be concise and medically relevant."
    )

    if patient_info:
        instruction += f" Patient info: {patient_info}."

    with session_lock:
        conversation_sessions[session_id] = [
            {"role": "user", "content": instruction},
            {"role": "model", "content": "I understand. Please describe your symptoms."}
        ]

    print(f"✅ Session started: {session_id}")

    return jsonify({
        "session_id": session_id,
        "message": "I understand. Please describe your symptoms.",
        "status": "success"
    })


# -------------------------
# Chat endpoint
# -------------------------
@app.route("/chat", methods=["POST"])
def chat():
    try:
        data = request.json or {}

        session_id = str(
            data.get("session_id") or data.get("userId") or "default"
        )
        user_message = data.get("message", "")
        max_tokens = int(data.get("max_tokens", 126))  # 126
        temperature = float(data.get("temperature", 0.2))

        if not user_message:
            return jsonify({"error": "Message is required"}), 400

        if model is None:
            return jsonify({"error": "Model not loaded"}), 503

        with session_lock:
            if session_id not in conversation_sessions:
                conversation_sessions[session_id] = [
                    {"role": "user", "content": "Instruction: You are a medical triage doctor."},
                    {"role": "model", "content": "I understand."}
                ]
            messages = conversation_sessions[session_id].copy()

        messages.append({"role": "user", "content": user_message})

        print(f"\n💬 [{session_id}] User: {user_message}")

        # ---- Prompt ----
        prompt = processor.apply_chat_template(
            messages,
            add_generation_prompt=True,
            tokenize=False
        )

        inputs = processor(
            text=prompt,
            return_tensors="pt"
        ).to(model.device)

        # ---- Generate ----
        with torch.no_grad():
            output = model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                do_sample=True,
                temperature=temperature,
                repetition_penalty=1.2,
                pad_token_id=processor.tokenizer.eos_token_id
            )

        # ---- SAFE response extraction (CRITICAL FIX) ----
        prompt_len = inputs.input_ids.shape[1]
        generated_tokens = output[0][prompt_len:]
        response_text = processor.tokenizer.decode(
            generated_tokens,
            skip_special_tokens=True
        ).strip()

        print(f"🤖 [{session_id}] AI: {response_text}")

        messages.append({"role": "model", "content": response_text})

        with session_lock:
            conversation_sessions[session_id] = messages

        return jsonify({
            "session_id": session_id,
            "response": response_text,
            "status": "success"
        })

    except Exception as e:
        print(f"❌ Chat error: {e}")
        return jsonify({"error": str(e)}), 500


# -------------------------
# End session
# -------------------------
@app.route("/end_session", methods=["POST"])
def end_session():
    data = request.json or {}
    session_id = str(data.get("session_id") or data.get("userId") or "default")

    with session_lock:
        conversation_sessions.pop(session_id, None)

    print(f"🔚 Session ended: {session_id}")

    return jsonify({"status": "success"})

# Root
