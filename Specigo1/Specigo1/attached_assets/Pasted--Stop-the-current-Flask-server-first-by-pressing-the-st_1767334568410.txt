from flask import Flask, request, jsonify
from flask_cors import CORS
from transformers import AutoProcessor, AutoModelForImageTextToText
import torch
from collections import defaultdict
import threading
from pyngrok import ngrok
import time

app = Flask(__name__)
CORS(app)

# Global variables
model = None
processor = None
conversation_sessions = defaultdict(list)
session_lock = threading.Lock()

print("=" * 70)
print("🏥 MEDGEMMA MEDICAL ASSISTANT API")
print("=" * 70)

# Use already loaded model from notebook
def setup_model():
    global model, processor

    # Try to use existing model first
    try:
        # Check if model exists in globals
        model_id = "google/medgemma-4b-it"
        if model is None:
            print("\n📦 Loading MedGemma model...")
            processor = AutoProcessor.from_pretrained(model_id)
            model = AutoModelForImageTextToText.from_pretrained(
                model_id,
                device_map="auto",
                quantization_config={"load_in_4bit": True},
                torch_dtype=torch.bfloat16,
            )
        print("✅ Model ready!")
        print(f"🎯 Device: {model.device}")
        return True
    except Exception as e:
        print(f"❌ Model error: {e}")
        return False

# API Routes
@app.route('/health', methods=['GET'])
def health_check():
    return jsonify({
        "status": "healthy",
        "model_loaded": model is not None,
        "active_sessions": len(conversation_sessions),
        "device": str(model.device) if model else "N/A"
    })

@app.route('/start_session', methods=['POST'])
def start_session():
    try:
        data = request.json
        session_id = data.get('session_id', 'default')
        patient_info = data.get('patient_info', '')

        instruction = "Instruction: You are a triage doctor. Ask ONE follow-up question per turn. Be concise and focused."
        if patient_info:
            instruction += f" Consider patient info: {patient_info}"
        instruction += " Do not repeat these instructions."

        with session_lock:
            conversation_sessions[session_id] = [
                {"role": "user", "content": instruction},
                {"role": "model", "content": "I understand. I will conduct a clinical interview. Please describe your symptoms."}
            ]

        print(f"✅ New session: {session_id}")
        return jsonify({
            "session_id": session_id,
            "message": "I understand. I will conduct a clinical interview. Please describe your symptoms.",
            "status": "success"
        })
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route('/chat', methods=['POST'])
def chat():
    try:
        data = request.json
        session_id = data.get('session_id', 'default')
        user_message = data.get('message', '')
        max_tokens = data.get('max_tokens', 50)
        temperature = data.get('temperature', 0.2)

        if not user_message:
            return jsonify({"error": "No message provided"}), 400

        if model is None:
            return jsonify({"error": "Model not loaded"}), 503

        with session_lock:
            if session_id not in conversation_sessions:
                conversation_sessions[session_id] = [
                    {"role": "user", "content": "Instruction: You are a triage doctor."},
                    {"role": "model", "content": "I understand."}
                ]
            messages = conversation_sessions[session_id].copy()

        messages.append({"role": "user", "content": user_message})

        print(f"\n💬 [{session_id[:8]}] User: {user_message[:50]}...")

        # Generate response
        prompt = processor.apply_chat_template(messages, add_generation_prompt=True, tokenize=False)
        inputs = processor(text=prompt, return_tensors="pt").to(model.device)

        with torch.no_grad():
            output = model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                do_sample=True,
                temperature=temperature,
                repetition_penalty=1.2,
                pad_token_id=processor.tokenizer.eos_token_id
            )

        full_text = processor.decode(output[0], skip_special_tokens=True)
        new_response = full_text.split("model")[-1].strip()

        if "Instruction:" in new_response:
            new_response = new_response.split("Instruction:")[0].strip()

        lines = new_response.split('\n')
        new_response = lines[0].strip() if lines else new_response

        print(f"🤖 [{session_id[:8]}] AI: {new_response[:50]}...")

        messages.append({"role": "model", "content": new_response})
        with session_lock:
            conversation_sessions[session_id] = messages

        return jsonify({
            "session_id": session_id,
            "response": new_response,
            "question_count": len([m for m in messages if m["role"] == "model"]) - 1,
            "status": "success"
        })
    except Exception as e:
        print(f"❌ Error: {e}")
        return jsonify({"error": str(e)}), 500

@app.route('/end_session', methods=['POST'])
def end_session():
    try:
        session_id = request.json.get('session_id', 'default')
        with session_lock:
            if session_id in conversation_sessions:
                del conversation_sessions[session_id]
                print(f"🔚 Session ended: {session_id}")
        return jsonify({"message": "Session ended", "status": "success"})
    except Exception as e:
        return jsonify({"error": str(e)}), 500
